[edit] Default constructors, and constructors taking more than one element

Can we give an example such as the following, which is clearer about the issues of default constructor, and constructor with multiple elements:

Run this code
#include <iostream>
#include <map>
#include <string>
 
class A {
public:
    A(unsigned ii = 0, const std::string &s = "") : i(ii), str(s) { }
 
    unsigned    i;
    std::string str;
};
 
std::ostream &operator<<(std::ostream &os, const A &a)
{
    return os << a.i << '\t' << a.str;
}
 
int main()
{
    std::map<unsigned, A> m;
    m.emplace(1,    1);                          // constructor taking 1 argument
    m.emplace(std::piecewise_construct,
              std::forward_as_tuple(0),
              std::forward_as_tuple());          // default constructor
    m.emplace(std::piecewise_construct,
              std::forward_as_tuple(2),
              std::forward_as_tuple(2, "abc"));  // constructor taking 2 arguments
 
    for (const auto &p : m) {
        std::cout << p.first << ": \t" << p.second << std::endl;
    }
    return 0;
}

[edit] Making constructor calls visible? Different key and value type?

I was wondering whether we should replace the string with two different user-defined types which emit logs of their various constructor calls, something like

Run this code
#include <iostream>
#include <utility>
#include <string>
#include <map>
 
struct KeyT
{
  int mI;
  KeyT(int i) : mI(i)
  {
    std::cout << __FUNCTION__ << "(int)\n";
  }
  KeyT(int i1, int i2, int i3) : mI(i1+i2+i3)
  {
    std::cout << __FUNCTION__ << "(int, int, int)\n";
  }
  KeyT(const KeyT& rhs): mI(rhs.mI)
  {
    std::cout << __FUNCTION__ << "(const KeyT&)\n";
  }
  KeyT(KeyT&& rhs) noexcept : mI(rhs.mI)
  {
    std::cout << __FUNCTION__ << "(KeyT &&)\n";
  }
 
  auto operator<=>(const KeyT&) const = default;
};
 
auto& operator<<(std::ostream& os, const KeyT& k)
{
  return os << k.mI;
}
 
struct ValueT
{
  const char *mS;
  ValueT(const char *s): mS(s)
  {
    std::cout << __FUNCTION__ << "(const char *s)\n";
  }
  ValueT(const ValueT& rhs): mS(rhs.mS)
  {
    std::cout << __FUNCTION__ << "(const ValueT &)\n";
  }
  ValueT(ValueT&& rhs) noexcept : mS(rhs.mS)
  {
    std::cout << __FUNCTION__ << "(ValueT &&)\n";
  }
};
 
auto& operator<<(std::ostream& os, const ValueT& v)
{
  return os << (v.mS ? v.mS : "nullptr");
}
 
int main()
{
    std::map<KeyT, ValueT> kvMap;
 
    // uses pair's move constructor
    kvMap.emplace(std::make_pair(KeyT(0), ValueT("zero")));
    std::cout << "\n";
 
    // uses pair's converting move constructor
    kvMap.emplace(std::make_pair(1, "one"));
    std::cout << "\n";
 
    // uses pair's template constructor
    kvMap.emplace(2, "two");
    std::cout << "\n";
 
    // uses pair's piecewise constructor
    kvMap.emplace(std::piecewise_construct,
      std::forward_as_tuple(10,11,12),
      std::forward_as_tuple("three"));
    std::cout << "\n";
 
    for (const auto& p : kvMap) {
      std::cout << p.first << " => " << p.second << '\n';
    }
}

The output is 

ValueT::ValueT(const char *s)
KeyT::KeyT(int)
KeyT::KeyT(KeyT &&)
ValueT::ValueT(ValueT &&)
KeyT::KeyT(KeyT &&)
ValueT::ValueT(ValueT &&)

KeyT::KeyT(int)
ValueT::ValueT(const char *s)

KeyT::KeyT(int)
ValueT::ValueT(const char *s)

KeyT::KeyT(int, int, int)
ValueT::ValueT(const char *s)

0 => zero
1 => one
2 => two
33 => three

Think it's a bit verbose tbh. How about
Run this code
#include <iostream>
#include <utility>
#include <string>
#include <map>
 
int main()
{
    std::map<std::string, std::string> m;
 
    // uses pair's move constructor
    // the two strings are moved into a pair temporary, which is then moved into map storage
    m.emplace(std::make_pair(std::string("a"), std::string("a")));
 
    // uses pair's converting move constructor
    // a pair temporary is made from these arguments, which is then used to construct a string pair in map storage
    m.emplace(std::make_pair("b", "abcd"));
 
    // uses pair's template constructor
    // these arguments are used to construct a string pair in map storage
    m.emplace("d", "ddd");
 
    // uses pair's piecewise constructor
    // these tuples of arguments are used to construct a string pair in map storage
    m.emplace(std::piecewise_construct,
              std::forward_as_tuple("c"),
              std::forward_as_tuple(10, 'c'));
    // as of C++17, m.try_emplace("c", 10, 'c'); can be used
 
    for (const auto &p : m) {
        std::cout << p.first << " => " << p.second << '\n';
    }
}

Output: 

a => a
b => abcd
c => cccccccccc
d => ddd
Although even this feels like we're just explaining how forwarding works. --Ybab321 (talk) 08:25, 3 August 2023 (PDT)
